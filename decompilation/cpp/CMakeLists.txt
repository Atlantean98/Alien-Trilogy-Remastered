cmake_minimum_required(VERSION 3.20)
project(TrilogyRecovered LANGUAGES CXX)
# bgfx's sources have very long paths; deep folders can pass Windows' 260-character limit (fetched with core.longpaths,
# but some tools still struggle). Say so plainly instead of failing later with a missing-file compiler error.
string(LENGTH "${CMAKE_BINARY_DIR}" TRILOGY_BUILD_PATH_LENGTH)
if(WIN32 AND TRILOGY_BUILD_PATH_LENGTH GREATER 100)
    message(WARNING "The build folder's path is ${TRILOGY_BUILD_PATH_LENGTH} characters long. If the build fails with a missing file, "
                    "move the repository to a short folder such as C:/src/Alien-Trilogy-Remastered and configure again.")
endif()
# Keep this machine's folders out of the binaries: source, build and toolchain paths that the compiler records
# (debug info, lambda names, header asserts) are rewritten to short relative names.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    get_filename_component(TRILOGY_TOOLCHAIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(TRILOGY_TOOLCHAIN "${TRILOGY_TOOLCHAIN}" DIRECTORY)
    add_compile_options("-ffile-prefix-map=${CMAKE_BINARY_DIR}=build" "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=." "-ffile-prefix-map=${TRILOGY_TOOLCHAIN}=toolchain")
    add_compile_options($<$<CONFIG:Release>:-gline-tables-only>)   # line tables to decode crash offsets; packaging strips the shipped copy and keeps this one
endif()
add_library(trilogy_recovered STATIC OriginalUtilities.cpp InputRepeat.cpp RuntimeCallbacks.cpp TaskScheduler.cpp PlayerMovement.cpp WorldMovement.cpp PlayerStates.cpp PlayerControl.cpp WeaponRuntime.cpp ProjectileRuntime.cpp ProjectileSimulation.cpp CombatRuntime.cpp PlayerDamage.cpp DropRendering.cpp PickupRuntime.cpp PickupScene.cpp SoftwareRendering.cpp RenderCommands.cpp PolygonRasterizer.cpp ShaderSpans.cpp LevelLoading.cpp EnemyRuntime.cpp EnemyConstruction.cpp EnemyAI.cpp EnemySpecialTasks.cpp EnemyTask.cpp EnemyMovement.cpp DoorRuntime.cpp)
target_compile_features(trilogy_recovered PUBLIC cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(trilogy_recovered PRIVATE -O2 -fwrapv)
endif()
target_include_directories(trilogy_recovered PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(recovered_checks tests.cpp)
target_link_libraries(recovered_checks PRIVATE trilogy_recovered)
add_test(NAME recovered_checks COMMAND recovered_checks)
if(WIN32)
    add_library(behavior_probe SHARED behavior_probe.cpp)
    target_link_libraries(behavior_probe PRIVATE trilogy_recovered)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    set_source_files_properties(PolygonRasterizer.cpp PROPERTIES COMPILE_OPTIONS "-fwrapv")
else()
    message(FATAL_ERROR "Polygon recovery requires a compiler supporting -fwrapv")
endif()

if(WIN32)
    set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(BGFX_BUILD_TOOLS_GEOMETRY OFF CACHE BOOL "" FORCE)
    set(BGFX_BUILD_TOOLS_TEXTURE OFF CACHE BOOL "" FORCE)
    set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
    set(BGFX_CONFIG_VIDEO OFF CACHE BOOL "" FORCE)
    add_subdirectory(../third_party/bgfx.cmake ${CMAKE_BINARY_DIR}/bgfx)
    target_compile_options(bgfx PRIVATE -O2)
    target_compile_options(bx PRIVATE -O2)
    target_compile_options(bimg PRIVATE -O2)
    add_executable(recovered_level_explorer WIN32 LevelExplorer.cpp ExplorerPs1Memory.cpp ExplorerCoop.cpp ExplorerController.cpp ExplorerGpu.cpp ExplorerEnemies.cpp ExplorerAssets.cpp ExplorerAudio.cpp ExplorerHud.cpp ExplorerLogic.cpp ExplorerLighting.cpp ExplorerPickups.cpp ExplorerObjects.cpp ExplorerMission.cpp ExplorerMenu.cpp ExplorerPresentation.cpp)
    set_property(SOURCE LevelExplorer.cpp APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/CoopSecureTransport.cs;${CMAKE_CURRENT_SOURCE_DIR}/../../tools/build_coop_transport.ps1")
    set(SDL3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/sdl3/SDL3-3.4.20/x86_64-w64-mingw32")
    target_include_directories(recovered_level_explorer PRIVATE "${SDL3_ROOT}/include")
    target_link_libraries(recovered_level_explorer PRIVATE "${SDL3_ROOT}/lib/libSDL3.dll.a")
    add_custom_command(TARGET recovered_level_explorer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL3_ROOT}/bin/SDL3.dll" "$<TARGET_FILE_DIR:recovered_level_explorer>/SDL3.dll")
    add_custom_command(TARGET recovered_level_explorer POST_BUILD COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/build_coop_transport.ps1" -OutputDirectory "$<TARGET_FILE_DIR:recovered_level_explorer>")
    set_target_properties(recovered_level_explorer PROPERTIES OUTPUT_NAME recovered_world_explorer)
    FetchContent_GetProperties(bgfx SOURCE_DIR TRILOGY_BGFX_SOURCE)
    include(../third_party/bgfx.cmake/cmake/bgfxToolUtils.cmake)
    bgfx_compile_shaders(TYPE VERTEX SHADERS shaders/vs_world.sc VARYING_DEF ${CMAKE_CURRENT_SOURCE_DIR}/shaders/varying.def.sc INCLUDE_DIRS ${TRILOGY_BGFX_SOURCE}/src PROFILES s_5_0 spirv OUTPUT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/build/shaders OUT_FILES_VAR GPU_VS)
    bgfx_compile_shaders(TYPE FRAGMENT SHADERS shaders/fs_world.sc VARYING_DEF ${CMAKE_CURRENT_SOURCE_DIR}/shaders/varying.def.sc INCLUDE_DIRS ${TRILOGY_BGFX_SOURCE}/src PROFILES s_5_0 spirv OUTPUT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/build/shaders OUT_FILES_VAR GPU_FS)
    bgfx_compile_shaders(TYPE FRAGMENT SHADERS shaders/fs_photo_depth.sc shaders/fs_photo_dof.sc VARYING_DEF ${CMAKE_CURRENT_SOURCE_DIR}/shaders/varying.def.sc INCLUDE_DIRS ${TRILOGY_BGFX_SOURCE}/src PROFILES s_5_0 spirv OUTPUT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/build/shaders OUT_FILES_VAR PHOTO_FS)
    target_sources(recovered_level_explorer PRIVATE ${GPU_VS} ${GPU_FS} ${PHOTO_FS})
    target_compile_options(recovered_level_explorer PRIVATE -O2)
    target_link_libraries(recovered_level_explorer PRIVATE trilogy_recovered bgfx ws2_32 advapi32 user32 gdi32 shell32 msimg32 ole32 uuid avrt)
endif()
